function normaliseData(data) { const isValidFixture = match => { if (!match || typeof match !== "object") return false; const home = typeof match.home === "string" ? match.home : match.home?.name || match.homeTeam?.name || match.homeTeam; const away = typeof match.away === "string" ? match.away : match.away?.name || match.awayTeam?.name || match.awayTeam; const league = match.leagueName || match.league || match.competition || match.league_name; const kickoff = match.utcTime || match.kickoff || match.date || match.fixtureDate || match.fixture?.date; if (typeof home !== "string" || !home.trim()) return false; if (typeof away !== "string" || !away.trim()) return false; if (/^(home team|away team)$/i.test(home.trim())) return false; if (/^(home team|away team)$/i.test(away.trim())) return false; if (home.trim().toLowerCase() === away.trim().toLowerCase()) return false; if (typeof league !== "string" || !league.trim()) return false; if (!kickoff || Number.isNaN(Date.parse(kickoff))) return false; return true; }; const clean = matches => matches.filter(isValidFixture); if (Array.isArray(data)) { return { best: [], standard: clean(data) }; } if (!data || typeof data !== "object") { throw new Error("Invalid predict.json format."); } const best = clean( data.best_of_7 || data.best7 || data.bestOf7 || data.top_picks || [] ); const standard = clean( data.standard_18 || data.standard || data.fixtures || data.matches || [] ); const bestIds = new Set(best.map(m => String(m.fixtureId ?? m.fixture_id ?? m.id ?? `${m.home}-${m.away}-${m.utcTime || m.date}`) )); return { best, standard: standard.filter(m => !bestIds.has( String(m.fixtureId ?? m.fixture_id ?? m.id ?? `${m.home}-${m.away}-${m.utcTime || m.date}`) )) }; }
